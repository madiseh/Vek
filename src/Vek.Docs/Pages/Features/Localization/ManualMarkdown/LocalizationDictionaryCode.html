using System;
using System.Collections.Generic;
using System.Threading;
using Microsoft.Extensions.Localization;
using Vek;

internal class DictionaryVekLocalizer : VekLocalizer
{
    private Dictionary<string, string> _localization;

    public DictionaryVekLocalizer()
    {
        _localization = new()
        {
            { "VekDataGrid_IsEmpty", "ist leer" },
            { "VekDataGrid_IsNotEmpty", "ist nicht leer" },
            { "VekDataGrid_Contains", "enthält" },
            { "VekDataGrid_NotContains", "enthält nicht" },
        };
    }

    public override LocalizedString this[string key]
    {
        get
        {
            var currentCulture = Thread.CurrentThread.CurrentUICulture.Parent.TwoLetterISOLanguageName;
            if (currentCulture.Equals("de", StringComparison.InvariantCultureIgnoreCase)
                && _localization.TryGetValue(key, out var res))
            {
                return new(key, res);
            }
            else
            {
                return new(key, key, true);
            }
        }
    }
}
